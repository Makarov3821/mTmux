cmake_minimum_required(VERSION 3.22.1)
project(mtmux_voice LANGUAGES C CXX)
if(NOT EXISTS "${WHISPER_CPP_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "whisper.cpp missing; run scripts/fetch_voice_benchmark.py deps")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_IS_DEV OFF CACHE BOOL "" FORCE)
set(WHISPER_CURL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
add_subdirectory("${WHISPER_CPP_DIR}" whisper EXCLUDE_FROM_ALL)
add_library(mtmux_voice SHARED voice_jni.cpp)
target_compile_features(mtmux_voice PRIVATE cxx_std_17)
target_compile_options(mtmux_voice PRIVATE -O3 -fvisibility=hidden -fvisibility-inlines-hidden)
target_link_options(mtmux_voice PRIVATE -Wl,-z,max-page-size=16384 -Wl,--gc-sections)
target_link_libraries(mtmux_voice PRIVATE whisper log android)
